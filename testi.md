```
SELECT
    DATE_TRUNC('MONTH', DATEADD(MONTH, -1, CURRENT_DATE())) AS KEY_MONTH,
    KAYTTOPAIKKA_ID,
    COUNT(*) AS KUUKAUDEN_MITTAUKSET,
    COUNT_IF(PALUULAMPOTILA >= 65) AS KUUKAUDEN_YLITYKSET,
    ROUND(
        COUNT_IF(PALUULAMPOTILA >= 65) * 100.0
        / NULLIF(COUNT(*), 0),
        2
    ) AS KUUKAUDEN_YLITYS_PCT
FROM MITTAUKSET
WHERE MITTAUSAIKA >= DATE_TRUNC('MONTH', DATEADD(MONTH, -1, CURRENT_DATE()))
  AND MITTAUSAIKA < DATE_TRUNC('MONTH', CURRENT_DATE())
GROUP BY
    KAYTTOPAIKKA_ID;
```
